==================
Quoted scalars retain DEL, C1, and noncharacters as normal text
==================
["ab", 'ab', "￾"]
------------------

(stream
  (document
    (flow_sequence
      (flow_sequence_open)
      (double_quoted_scalar
        (quote_open)
        (scalar_text)
        (quote_close))
      (flow_separator)
      (separation)
      (single_quoted_scalar
        (quote_open)
        (scalar_text)
        (quote_close))
      (flow_separator)
      (separation)
      (double_quoted_scalar
        (quote_open)
        (scalar_text)
        (quote_close))
      (flow_sequence_close))))
